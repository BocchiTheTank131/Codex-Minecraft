cmake_minimum_required(VERSION 3.20)
project(StabilityDriver LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
get_filename_component(ROOT "../.." ABSOLUTE)
set(BASE "${ROOT}/build/installed-release" CACHE PATH "Matching production Release build")
file(GLOB OBJECTS "${BASE}/VoxelFrontier.dir/Release/*.obj")
list(FILTER OBJECTS EXCLUDE REGEX "/main.obj$")
add_executable(StabilityDriver main.cpp farming.cpp building.cpp FarmingBaseline.cpp ${OBJECTS})
target_include_directories(StabilityDriver PRIVATE "${ROOT}/include" "${BASE}/_deps/glm-src" "${BASE}/_deps/glfw-src/include" "${BASE}/_deps/glad-build/include")
target_compile_definitions(StabilityDriver PRIVATE VOXEL_STANDALONE NOMINMAX)
target_link_libraries(StabilityDriver PRIVATE "${BASE}/_deps/glfw-build/src/Release/glfw3.lib" "${BASE}/_deps/glad-build/Release/glad.lib" "${BASE}/_deps/glm-build/glm/Release/glm.lib" opengl32 windowscodecs ole32 psapi shell32 user32 gdi32 winmm advapi32)
target_sources(StabilityDriver PRIVATE "${BASE}/VoxelFrontier.dir/Release/VoxelResources.res")
target_link_options(StabilityDriver PRIVATE "${BASE}/VoxelFrontier.dir/Release/VoxelResources.res")
